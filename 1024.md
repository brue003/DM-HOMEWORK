```
N,K=map(int,input().split())
table=list(input())
count=0

for i in range(N):
    if table[i] == 'P':
        start=max(0,i-K)
        end=min(N-1,i+K)

        for j in range(start, end+1):
            if table[j]=="H":
                table[j]='0'
                count+=1
                break
    

print(count)
```

```
import sys
input = sys.stdin.readline
fridge=[]
N=int(input())
for i in range(N):
    x,y = map(int, input().split())
    fridge.append([x,y])

fridge.sort(key=lambda x: (x[1],x[0]))
tem=fridge[0][1]
count=1
for i in range(1, len(fridge)):
    if fridge[i][0] > tem:
        count+=1
        tem=fridge[i][1]
print(count)
```
